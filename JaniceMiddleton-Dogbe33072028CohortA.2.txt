CREATE VIEW EmpProject(empNo, projNo, totalHours)
AS SELECT w.empNo, w.projNo, SUM(hoursWorked)
   FROM   Employee e, Project p, WorksOn w
   WHERE  e.empNo = w.empNo AND p.projNo = w.projNo
   GROUP BY w.empNo, w.projNo;
